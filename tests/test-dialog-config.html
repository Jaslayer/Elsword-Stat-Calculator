📋 對話框元素 data-config 屬性驗證

測試指南

  1. 打開主應用 (index.html)
  2. 選擇一些統計項目（例如 "物/魔攻擊力" 和 "適應力"）
  3. 點擊 config1 和 config2 的按鈕打開對話框
  4. 在瀏覽器控制台(F12)執行測試腳本
  5. 查看結果

測試腳本 - 複製到控制台執行


// ========== 測試 1: 簡單項目對話框 (例如物/魔攻擊力) ==========
console.log('=== 測試簡單項目對話框 ===');
const stat0Dialog = document.querySelector('#stat-dialog-0');
if (stat0Dialog && stat0Dialog.classList.contains('show')) {
  const inputs = stat0Dialog.querySelectorAll('input[type="number"]');
  let allHaveConfig = true;
  inputs.forEach(input => {
    if (!input.dataset.config) {
      console.warn('❌ 找到無 data-config 的輸入框:', input.id);
      allHaveConfig = false;
    } else {
      console.log(`✅ ${input.className} 有 data-config="${input.dataset.config}"`);
    }
  });
  if (allHaveConfig) console.log('✅ 所有簡單項目輸入框都有 data-config');
} else {
  console.log('⚠️ 簡單項目對話框未打開，請先打開');
}

// ========== 測試 2: 適應力對話框 ==========
console.log('\n=== 測試適應力對話框 ===');
const adaptDialog = document.querySelector('#adaptabilityInputDialog');
if (adaptDialog && adaptDialog.classList.contains('show')) {
  const inputs = adaptDialog.querySelectorAll('input[type="number"], input[type="checkbox"], select');
  let allHaveConfig = true;
  inputs.forEach(input => {
    if (!input.dataset.config) {
      console.warn('❌ 找到無 data-config 的元素:', input.className);
      allHaveConfig = false;
    }
  });
  const checkboxes = adaptDialog.querySelectorAll('input[type="checkbox"]');
  checkboxes.forEach(cb => {
    console.log(`✅ 複選框 ${cb.className} 有 data-config="${cb.dataset.config}"`);
  });
  if (allHaveConfig) console.log('✅ 所有適應力對話框元素都有 data-config');
} else {
  console.log('⚠️ 適應力對話框未打開，請先打開');
}

// ========== 測試 3: 致命一擊傷害對話框 ==========
console.log('\n=== 測試致命一擊傷害對話框 ===');
const critDialog = document.querySelector('#criticalDamageInputDialog');
if (critDialog && critDialog.classList.contains('show')) {
  const panelInput = critDialog.querySelector('.dialog-panel-input');
  console.log(`✅ 面板輸入框 有 data-config="${panelInput.dataset.config}"`);
  
  const additives = critDialog.querySelectorAll('.dialog-additive-input');
  additives.forEach((input, idx) => {
    if (!input.dataset.config) {
      console.warn(`❌ 加成傷害 #${idx} 無 data-config`);
    } else {
      console.log(`✅ 加成傷害輸入框 有 data-config="${input.dataset.config}"`);
    }
  });
  
  const multiplicatives = critDialog.querySelectorAll('.dialog-multiplicative-input');
  multiplicatives.forEach((input, idx) => {
    if (!input.dataset.config) {
      console.warn(`❌ 乘算爆傷 #${idx} 無 data-config`);
    } else {
      console.log(`✅ 乘算爆傷輸入框 有 data-config="${input.dataset.config}"`);
    }
  });
  
  if (additives.length === 0 && multiplicatives.length === 0) {
    console.log('⚠️ 對話框中沒有動態行，請先添加加成傷害或乘算爆傷');
  }
} else {
  console.log('⚠️ 致命一擊傷害對話框未打開，請先打開');
}

console.log('\n=== 測試完成 ===');

預期結果

  • ✅ 所有輸入框（input type="number"）都有 data-config="1" 或 data-config="2"
  • ✅ 所有複選框（input type="checkbox"）都有 data-config 屬性
  • ✅ 所有下拉選單（select）都有 data-config 屬性
  • ✅ 預設為 data-config 的值對應於打開對話框的按鈕的配置號
  • ✅ 動態添加的新行（加成傷害、乘算爆傷）也都有 data-config 屬性
